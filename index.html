function doPost(e) { try { // --- CANDADO DE FECHA Y HORA DE CIERRE --- // Viernes 7 de Agosto de 2026 a las 21:00 Hrs (Mes 7 es Agosto en JS) var fechaCierre = new Date(2026, 7, 7, 21, 0, 0); var fechaActual = new Date(); if (fechaActual > fechaCierre) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "La quiniela ya cerró el viernes 7 de agosto a las 21:00 hrs. ¡Suerte para la próxima!" })).setMimeType(ContentService.MimeType.JSON); } // ----------------------------------------- var datos = JSON.parse(e.postData.contents); var nombreInput = datos.nombre; var nombreBuscar = nombreInput.toString().trim().toLowerCase(); var pronosticos = datos.quiniela[0]; var libro = SpreadsheetApp.getActiveSpreadsheet(); var hoja = libro.getSheetByName("QUINIELAS"); if (!hoja) { hoja = libro.getSheets()[0]; } // --- CANDADO PARA NOMBRES REPETIDOS --- var ultimaFila = hoja.getLastRow(); if (ultimaFila >= 2) { var nombresExistentes = hoja.getRange(2, 1, ultimaFila - 1, 1).getValues(); for (var j = 0; j < nombresExistentes.length; j++) { var nombreActual = nombresExistentes[j][0].toString().trim().toLowerCase(); if (nombreActual === nombreBuscar) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": "Este nombre ya está registrado. Por favor usa otro." })).setMimeType(ContentService.MimeType.JSON); } } } // -------------------------------------- // Organizar la fila para los 9 partidos var nuevaFila = [nombreInput]; for (var i = 0; i < 9; i++) { nuevaFila.push(pronosticos[i] || ""); } // Encontrar la primera fila vacía en la Columna A var valoresA = hoja.getRange("A1:A105").getValues(); var filaVacia = 2; for (var k = 1; k < valoresA.length; k++) { if (valoresA[k][0] === "") { filaVacia = k + 1; break; } } hoja.getRange(filaVacia, 1, 1, nuevaFila.length).setValues([nuevaFila]); return ContentService.createTextOutput(JSON.stringify({ "status": "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { var libro = SpreadsheetApp.getActiveSpreadsheet(); var hoja = libro.getSheetByName("QUINIELAS"); if (!hoja) hoja = libro.getSheets()[0]; var ultimaFila = hoja.getLastRow(); var nombres = []; if (ultimaFila >= 2) { var datos = hoja.getRange(2, 1, ultimaFila - 1, 1).getValues(); for (var i = 0; i < datos.length; i++) { if (datos[i][0] !== "") { nombres.push(datos[i][0].toString().trim().toLowerCase()); } } } return ContentService.createTextOutput(JSON.stringify(nombres)) .setMimeType(ContentService.MimeType.JSON); }
